cmake_minimum_required(VERSION 3.20)

# Set vcpkg toolchain file BEFORE project()
set(CMAKE_TOOLCHAIN_FILE "${CMAKE_SOURCE_DIR}/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")

project(GameEngine)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Collect all source files from GameEngine/Source directory
file(GLOB_RECURSE SOURCES 
    "GameEngine/Source/Common/*.cpp"
    "GameEngine/Source/Compute/*.cpp"

    "GameEngine/Source/GFX/*.cpp"
    "GameEngine/Source/ImGui/*.cpp"
    "GameEngine/Source/Physics/*.cpp"

    "GameEngine/Source/Common/*.c"
    "GameEngine/Source/Compute/*.c"
    "GameEngine/Source/ECS/*.c"
    "GameEngine/Source/GFX/*.c"
    "GameEngine/Source/ImGui/*.c"
    "GameEngine/Source/Physics/*.c"
)

    "GameEngine/Source/Common/*.h"
    "GameEngine/Source/Compute/*.h"

    "GameEngine/Source/GFX/*.h"
    "GameEngine/Source/ImGui/*.h"

    "GameEngine/Source/GFX/*.hpp"

)

# Create executable
add_executable(GameEngine WIN32 ${SOURCES} ${HEADERS})
# The target name is: GameEngine

# 1. Target the 'GameEngine' directory relative to the source root
# Define the source code directory
set(ASSET_ROOT_DIR "${CMAKE_SOURCE_DIR}/GameEngine")

# 2. Set the property on your executable target
set_target_properties(GameEngine PROPERTIES 
                        
    # Use the variable defined above to set the working directory for the VS debugger
    VS_DEBUGGER_WORKING_DIRECTORY "${ASSET_ROOT_DIR}"
)

# Include directories
target_include_directories(GameEngine PRIVATE
    # Source directories
    ${CMAKE_SOURCE_DIR}/GameEngine/Source
    ${CMAKE_SOURCE_DIR}/GameEngine/Source/Common
    ${CMAKE_SOURCE_DIR}/GameEngine/Source/Compute
    ${CMAKE_SOURCE_DIR}/GameEngine/Source/ECS
    ${CMAKE_SOURCE_DIR}/GameEngine/Source/GFX
    ${CMAKE_SOURCE_DIR}/GameEngine/Source/ImGui
    ${CMAKE_SOURCE_DIR}/GameEngine/Source/Physics
    
    # External includes
    ${CMAKE_SOURCE_DIR}/Includes
    ${CMAKE_SOURCE_DIR}/Includes/entt
    ${CMAKE_SOURCE_DIR}/Includes/PhysX
    
    # vcpkg DirectX includes
    ${CMAKE_SOURCE_DIR}/vcpkg_installed/x64-windows/include/directx
)

# Compile definitions
target_compile_definitions(GameEngine PRIVATE
    _CRT_SECURE_NO_WARNINGS
    DX12_API
    _SILENCE_CXX20_CISO646_REMOVED_WARNING
    PIX
    ENABLE_GPU_BASED_VALIDATION
    _CONSOLE
    NOMINMAX
    WIN32_LEAN_AND_MEAN
    UNICODE
    _UNICODE
)

# MSVC settings
if(MSVC)
    target_compile_options(GameEngine PRIVATE 
        /W3                 # Warning level 3
        /MP                 # Multi-processor compilation
        /permissive-        # Standards conformance
        /Zc:__cplusplus     # Enable correct __cplusplus macro
    )
    
    # Debug-specific flags
    target_compile_options(GameEngine PRIVATE 
        $<$<CONFIG:Debug>:/Od /Zi /RTC1>
    )
    
    # Release-specific flags
    target_compile_options(GameEngine PRIVATE 
        $<$<CONFIG:Release>:/O2 /GL>
    )
    
    # Link-time code generation for Release
    target_link_options(GameEngine PRIVATE
        $<$<CONFIG:Release>:/LTCG>
    )
endif()
